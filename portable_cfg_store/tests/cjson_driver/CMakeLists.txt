cmake_minimum_required(VERSION 3.16)
project(pcfg_external_cjson_tests C CXX)
enable_testing()
set(PCFG_CJSON_SOURCE_DIR "" CACHE PATH "External cJSON checkout")
if(NOT EXISTS "${PCFG_CJSON_SOURCE_DIR}/cJSON.h")
    message(FATAL_ERROR "Supply PCFG_CJSON_SOURCE_DIR pointing to external cJSON")
endif()
set(ENABLE_CJSON_TEST OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PCFG_CJSON_SOURCE_DIR}" external-cjson)
target_include_directories(cjson PUBLIC "${PCFG_CJSON_SOURCE_DIR}")
set(PCFG_BUILD_CJSON_ADAPTER ON CACHE BOOL "" FORCE)
add_subdirectory(../.. portable-cfg)
